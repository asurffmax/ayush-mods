/* ============================ rgb engine ============================ */

object Rgb {
    fun norm(d: Float): Float = ((d % 360f) + 360f) % 360f
    fun hue(d: Float, value: Float = 1f): Int =
        Color.HSVToColor(floatArrayOf(norm(d), 1f, value.coerceIn(0f, 1f)))
    fun rainbow(size: Int = 9): IntArray =
        IntArray(size) { hue(it * (360f / (size - 1))) }
}

/** FrameLayout that paints an animated rotating rainbow border. */
class RgbBorder @JvmOverloads constructor(
    context: Context, attrs: AttributeSet? = null
) : FrameLayout(context, attrs) {

    var cornerDp: Float = 18f
    private val stroke = context.dpf(2.4f)
    private val paint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        style = Paint.Style.STROKE
        strokeWidth = stroke
    }
    private val rect = RectF()
    private val mat = Matrix()
    private var shader: SweepGradient? = null
    private var deg = 0f

    init { setPadding(context.dpi(3f), context.dpi(3f), context.dpi(3f), context.dpi(3f)) }

    override fun onSizeChanged(w: Int, h: Int, ow: Int, oh: Int) {
        shader = SweepGradient(w / 2f, h / 2f, Rgb.rainbow(), null)
    }

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)
        val s = shader ?: return
        deg = (deg + 1.8f) % 360f
        mat.setRotate(deg, width / 2f, height / 2f)
        s.setLocalMatrix(mat)
        paint.shader = s
        val o = stroke / 2f
        rect.set(o, o, width - o, height - o)
        val r = context.dpf(cornerDp)
        canvas.drawRoundRect(rect, r, r, paint)
        postInvalidateOnAnimation()
    }
}

/** TextView whose text cycles through a rainbow gradient. */
class RgbText @JvmOverloads constructor(
    context: Context, attrs: AttributeSet? = null
) : TextView(context, attrs) {

    private var grad: LinearGradient? = null
    private var off = 0f

    init {
        includeFontPadding = false
        setTypeface(typeface, Typeface.BOLD)
    }

    override fun onSizeChanged(w: Int, h: Int, ow: Int, oh: Int) {
        if (w > 0) grad = LinearGradient(
            0f, 0f, w.toFloat(), 0f, Rgb.rainbow(), null, Shader.TileMode.REPEAT
        )
    }

    override fun onDraw(canvas: Canvas) {
        val g = grad
        if (g != null) {
            off = (off + 1.6f) % 10000f
            val m = Matrix()
            m.setTranslate(off, 0f)
            g.setLocalMatrix(m)
            paint.shader = g
            postInvalidateOnAnimation()
        }
        super.onDraw(canvas)
    }
}

/** Small hue-cycling status dot with soft glow. */
class RgbDot @JvmOverloads constructor(
    context: Context, attrs: AttributeSet? = null
) : View(context, attrs) {

    private val core = Paint(Paint.ANTI_ALIAS_FLAG)
    private val glow = Paint(Paint.ANTI_ALIAS_FLAG)
    private var deg = 0f

    override fun onMeasure(w: Int, h: Int) =
        setMeasuredDimension(context.dpi(9f), context.dpi(9f))

    override fun onDraw(canvas: Canvas) {
        deg = (deg + 3f) % 360f
        val c = Rgb.hue(deg)
        glow.color = Color.argb(75, Color.red(c), Color.green(c), Color.blue(c))
        core.color = c
        val r = min(width, height) / 2f
        canvas.drawCircle(width / 2f, height / 2f, r, glow)
        canvas.drawCircle(width / 2f, height / 2f, r * 0.58f, core)
        postInvalidateOnAnimation()
    }
}

/** Custom toggle with an RGB track when on. */
class RgbSwitch @JvmOverloads constructor(
    context: Context, attrs: AttributeSet? = null
) : View(context, attrs) {

    var onChanged: ((Boolean) -> Unit)? = null
    var isOn: Boolean = false
        private set

    private var prog = 0f
    private var deg = 0f
    private val track = Paint(Paint.ANTI_ALIAS_FLAG)
    private val knob = Paint(Paint.ANTI_ALIAS_FLAG)
    private val rect = RectF()

    init {
        isClickable = true
        isFocusable = true
    }

    override fun onMeasure(w: Int, h: Int) =
        setMeasuredDimension(context.dpi(46f), context.dpi(26f))

    fun setOn(on: Boolean, notify: Boolean = false) {
        if (isOn == on) return
        isOn = on
        ValueAnimator.ofFloat(prog, if (on) 1f else 0f).apply {
            duration = 200
            interpolator = DecelerateInterpolator()
            addUpdateListener { prog = it.animatedValue as Float; invalidate() }
            start()
        }
        if (notify) onChanged?.invoke(on)
    }

    override fun onTouchEvent(e: MotionEvent): Boolean = when (e.actionMasked) {
        MotionEvent.ACTION_DOWN -> { isPressed = true; true }
        MotionEvent.ACTION_UP -> {
            isPressed = false
            setOn(!isOn, true)
            performClick()
            true
        }
        MotionEvent.ACTION_CANCEL -> { isPressed = false; true }
        else -> super.onTouchEvent(e)
    }

    override fun onDraw(canvas: Canvas) {
        deg = (deg + 2.2f) % 360f
        val h = height.toFloat()
        val r = h / 2f
        rect.set(0f, 0f, width.toFloat(), h)

        if (prog > 0.02f) {
            track.shader = LinearGradient(
                0f, 0f, width.toFloat(), 0f,
                intArrayOf(Rgb.hue(deg), Rgb.hue(deg + 80f), Rgb.hue(deg + 160f)),
                null, Shader.TileMode.CLAMP
            )
            track.alpha = (45 + 145 * prog).toInt().coerceIn(0, 255)
        } else {
            track.shader = null
            track.color = 0xFF22302A.toInt()
        }
        canvas.drawRoundRect(rect, r, r, track)

        val kx = context.dpf(13f) + prog * (width - context.dpf(26f))
        knob.color = mix(C_DIM, Rgb.hue(deg + 40f), prog)
        canvas.drawCircle(kx, r, r - context.dpf(3f), knob)
        postInvalidateOnAnimation()
    }

    private fun mix(a: Int, b: Int, t: Float): Int = Color.argb(
        (Color.alpha(a) + (Color.alpha(b) - Color.alpha(a)) * t).toInt(),
        (Color.red(a) + (Color.red(b) - Color.red(a)) * t).toInt(),
        (Color.green(a) + (Color.green(b) - Color.green(a)) * t).toInt(),
        (Color.blue(a) + (Color.blue(b) - Color.blue(a)) * t).toInt()
    )
}

/** Indeterminate rainbow progress bar. */
class RgbProgress @JvmOverloads constructor(
    context: Context, attrs: AttributeSet? = null
) : View(context, attrs) {

    private val paint = Paint(Paint.ANTI_ALIAS_FLAG)
    private val rect = RectF()
    private var t = 0f

    override fun onMeasure(w: Int, h: Int) =
        setMeasuredDimension(context.dpi(180f), context.dpi(4f))

    override fun onDraw(canvas: Canvas) {
        val h = height.toFloat()
        val r = h / 2f
        paint.shader = null
        paint.color = 0xFF12201A.toInt()
        rect.set(0f, 0f, width.toFloat(), h)
        canvas.drawRoundRect(rect, r, r, paint)

        t = (t + 0.012f) % 1f
        val seg = width * 0.34f
        val x = -seg + (width + seg) * t
        rect.set(x, 0f, x + seg, h)
        paint.shader = LinearGradient(
            x, 0f, x + seg, 0f,
            intArrayOf(Rgb.hue(t * 720f), Rgb.hue(t * 720f + 100f)),
            null, Shader.TileMode.CLAMP
        )
        canvas.drawRoundRect(rect, r, r, paint)
        postInvalidateOnAnimation()
    }
}

/** Headshot reticle. RGB hue-cycling, head ring sits above centre. */
class HeadReticle @JvmOverloads constructor(
    context: Context, attrs: AttributeSet? = null
) : View(context, attrs) {

    private val d = resources.displayMetrics.density
    private val ring = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        style = Paint.Style.STROKE; strokeWidth = 1.5f * d
    }
    private val tick = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        style = Paint.Style.STROKE; strokeWidth = 1.5f * d
    }
    private val dot = Paint(Paint.ANTI_ALIAS_FLAG)

    private var deg = 0f
    private var phase = 0f

    override fun onMeasure(w: Int, h: Int) {
        val s = (64 * d).toInt()
        setMeasuredDimension(s, s)
    }

    override fun onDraw(canvas: Canvas) {
        deg = (deg + 0.9f) % 360f
        phase += 0.055f
        val alpha = (150 + 90 * sin(phase)).toInt().coerceIn(90, 255)
        val c = Rgb.hue(deg)

        ring.color = c; ring.alpha = alpha
        tick.color = c; tick.alpha = alpha
        dot.color = c

        val cx = width / 2f
        val cy = height / 2f
        val r = 26f * d
        val t = 6f * d

        canvas.drawCircle(cx, cy, r, ring)
        canvas.drawLine(cx, cy - r, cx, cy - r - t, tick)
        canvas.drawLine(cx, cy + r, cx, cy + r + t, tick)
        canvas.drawLine(cx - r, cy, cx - r - t, cy, tick)
        canvas.drawLine(cx + r, cy, cx + r + t, cy, tick)
        canvas.drawCircle(cx, cy - 7f * d, 9f * d, ring)
        canvas.drawCircle(cx, cy, 2.5f * d, dot)

        postInvalidateOnAnimation()
    }
}

        } else {
            setTextColor(0xFF8FFFC4.toInt())
            background = roundBg(ctx, 12f, 0xFF0C1410.toInt(), 0x3339FF14, 1f)
        }
    }

/* ============================ accessibility ============================ */

class PanelAccessibilityService : AccessibilityService() {

    override fun onServiceConnected() {
        instance = this
        serviceInfo = (serviceInfo ?: AccessibilityServiceInfo()).apply {
            eventTypes = AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED
            feedbackType = AccessibilityServiceInfo.FEEDBACK_GENERIC
            capabilities = AccessibilityServiceInfo.CAPABILITY_CAN_PERFORM_GESTURES
        }
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {}
    override fun onInterrupt() {}

    fun nudgeUp(px: Float) {
        val dm = resources.displayMetrics
        val x = dm.widthPixels / 2f
        val y = dm.heightPixels / 2f
        val path = Path().apply { moveTo(x, y); lineTo(x, y - px) }
        runCatching {
            dispatchGesture(
                GestureDescription.Builder()
                    .addStroke(GestureDescription.StrokeDescription(path, 0L, 70L))
                    .build(), null, null
            )
        }
    }

    companion object {
        var instance: PanelAccessibilityService? = null
    }
}

/* ============================ overlay service ============================ */

class OverlayService : Service() {

    private lateinit var wm: WindowManager
    private var panel: View? = null
    private var panelParams: WindowManager.LayoutParams? = null
    private var reticle: HeadReticle? = null
    private var bubble: View? = null
    private var bubbleParams: WindowManager.LayoutParams? = null

    private var swHead: RgbSwitch? = null
    private var swChams: RgbSwitch? = null
    private var tvHead: TextView? = null
    private var tvChams: TextView? = null
    private var panelInner: LinearLayout? = null

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onCreate() {
        super.onCreate()
        goForeground()
        wm = getSystemService(WINDOW_SERVICE) as WindowManager
        buildReticle()
        buildPanel()
        buildBubble()
        sync()
    }

    /* ---------- reticle ---------- */
    private fun buildReticle() {
        val v = HeadReticle(this)
        val p = WindowManager.LayoutParams(
            WRAP, WRAP, WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE or
                WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
            PixelFormat.TRANSLUCENT
        ).apply { gravity = Gravity.CENTER }
        reticle = v
        wm.addView(v, p)
    }

    /* ---------- panel ---------- */
    private fun buildPanel() {
        val ctx = this
        val border = RgbBorder(ctx).apply { cornerDp = 18f }
        val col = LinearLayout(ctx).apply {
            orientation = LinearLayout.VERTICAL
            background = roundBg(ctx, 15f, 0xF00A0F0C.toInt())
        }
        border.addView(col, FrameLayout.LayoutParams(MATCH, WRAP))
        panelInner = col

        /* header */
        val header = LinearLayout(ctx).apply {
            orientation = LinearLayout.HORIZONTAL
            gravity = Gravity.CENTER_VERTICAL
            setPadding(ctx.dpi(12f), 0, ctx.dpi(4f), 0)
        }
        header.addView(RgbDot(ctx), LinearLayout.LayoutParams(ctx.dpi(9f), ctx.dpi(9f)))
        header.addView(
            RgbText(ctx).apply {
                text = "AYUSH MODS"
                textSize = 12.5f
                letterSpacing = 0.16f
            },
            LinearLayout.LayoutParams(0, WRAP, 1f).apply { marginStart = ctx.dpi(8f) }
        )
        header.addView(
            txt(ctx, "–", 16f, 0xFF8FFFC4.toInt(), true).apply {
                gravity = Gravity.CENTER
                setOnClickListener { collapse() }
            },
            LinearLayout.LayoutParams(ctx.dpi(30f), ctx.dpi(30f))
        )
        header.addView(
            txt(ctx, "✕", 12f, 0xFF8FFFC4.toInt(), true).apply {
                gravity = Gravity.CENTER
                setOnClickListener { stopSelf() }
            },
            LinearLayout.LayoutParams(ctx.dpi(30f), ctx.dpi(30f))
        )
        col.addView(header, LinearLayout.LayoutParams(MATCH, ctx.dpi(42f)))
        col.addView(divider(ctx, C_DIV))

        /* body */
        val body = LinearLayout(ctx).apply {
            orientation = LinearLayout.VERTICAL
            setPadding(ctx.dpi(12f), ctx.dpi(12f), ctx.dpi(12f), ctx.dpi(12f))
        }
        col.addView(body, LinearLayout.LayoutParams(MATCH, WRAP))

        val sHead = RgbSwitch(ctx)
        val stHead = txt(ctx, "OFF", 10f, C_DIM, true)
        val sChams = RgbSwitch(ctx)
        val stChams = txt(ctx, "OFF", 10f, C_DIM, true)
        swHead = sHead; swChams = sChams
        tvHead = stHead; tvChams = stChams

        body.addView(row(ctx, "HEADSHOT", "Head lock aim", stHead, sHead),
            LinearLayout.LayoutParams(MATCH, WRAP))
        body.addView(row(ctx, "CHAMS", "Green highlight", stChams, sChams),
            LinearLayout.LayoutParams(MATCH, WRAP).apply { topMargin = ctx.dpi(14f) })

        /* footer */
        col.addView(divider(ctx, C_DIV2))
        val footer = LinearLayout(ctx).apply {
            orientation = LinearLayout.HORIZONTAL
            gravity = Gravity.CENTER_VERTICAL
            setPadding(ctx.dpi(12f), ctx.dpi(9f), ctx.dpi(12f), ctx.dpi(9f))
        }
        footer.addView(txt(ctx, "OB55 · NON-ROOT · RGB", 9f, 0xFF6F8B79.toInt()).apply {
            letterSpacing = 0.12f
        })
        footer.addView(View(ctx), LinearLayout.LayoutParams(0, 1, 1f))
        footer.addView(txt(ctx, "v3.0", 9f, 0xFF4E6B59.toInt()))
        col.addView(footer, LinearLayout.LayoutParams(MATCH, WRAP))

        sHead.onChanged = { apply(Prefs.HEADSHOT, it) }
        sChams.onChanged = { apply(Prefs.CHAMS, it) }

        val p = WindowManager.LayoutParams(
            ctx.dpi(248f), WRAP, WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS or
                WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.START
            x = Prefs.sp(ctx).getInt(Prefs.X, 32)
            y = Prefs.sp(ctx).getInt(Prefs.Y, 160)
        }
        panelParams = p
        panel = border
        wm.addView(border, p)

        drag(header, border, p, { savePos(p) }) {
            PanelAccessibilityService.instance?.nudgeUp(24f * resources.displayMetrics.density)
        }
    }

    private fun row(
        ctx: Context, title: String, sub: String, state: TextView, sw: RgbSwitch
    ): LinearLayout {
        val r = LinearLayout(ctx).apply {
            orientation = LinearLayout.HORIZONTAL
            gravity = Gravity.CENTER_VERTICAL
        }
        val left = LinearLayout(ctx).apply { orientation = LinearLayout.VERTICAL }
        left.addView(txt(ctx, title, 12.5f, C_TEXT, true).apply { letterSpacing = 0.1f })
        left.addView(txt(ctx, sub, 10f, C_SUB))
        r.addView(left, LinearLayout.LayoutParams(0, WRAP, 1f))
        r.addView(state, LinearLayout.LayoutParams(WRAP, WRAP).apply { marginEnd = ctx.dpi(8f) })
        r.addView(sw, LinearLayout.LayoutParams(WRAP, WRAP))
        return r
    }

    /* ---------- bubble ---------- */
    private fun buildBubble() {
        val ctx = this
        val border = RgbBorder(ctx).apply { cornerDp = 23f }
        border.addView(
            RgbText(ctx).apply {
                text = "A"
                textSize = 20f
                gravity = Gravity.CENTER
            },
            FrameLayout.LayoutParams(MATCH, MATCH)
        )
        val p = WindowManager.LayoutParams(
            ctx.dpi(46f), ctx.dpi(46f), WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.START
            x = Prefs.sp(ctx).getInt(Prefs.X, 32)
            y = Prefs.sp(ctx).getInt(Prefs.Y, 160)
        }
        bubble = border
        bubbleParams = p
        border.visibility = View.GONE
        wm.addView(border, p)

        var moved = false
        drag(border, border, p, { moved = true; savePos(p) }, null)
        border.setOnClickListener {
            if (moved) { moved = false; return@setOnClickListener }
            border.visibility = View.GONE
            panel?.visibility = View.VISIBLE
            panelParams?.let { pp ->
                pp.x = p.x; pp.y = p.y
                panel?.let { v -> runCatching { wm.updateViewLayout(v, pp) } }
            }
        }
    }

    private fun collapse() {
        val p = panelParams ?: return
        panel?.visibility = View.GONE
        bubbleParams?.apply { x = p.x; y = p.y }
        bubble?.visibility = View.VISIBLE
        savePos(p)
    }

    /* ---------- toggle ---------- */
    private fun apply(key: String, on: Boolean) {
        Prefs.set(this, key, on)
        when (key) {
            Prefs.HEADSHOT -> {
                reticle?.visibility = if (on) View.VISIBLE else View.GONE
                tvHead?.text = if (on) "HEAD" else "OFF"
                tvHead?.setTextColor(if (on) C_GREEN else C_DIM)
            }
            Prefs.CHAMS -> {
                tvChams?.text = if (on) "GREEN" else "OFF"
                tvChams?.setTextColor(if (on) C_GREEN else C_DIM)
                panelInner?.background = roundBg(
                    this, 15f,
                    if (on) 0xF00C1710.toInt() else 0xF00A0F0C.toInt()
                )
            }
        }
        writeBridge()
        sendBroadcast(
            Intent(ACTION_TOGGLE).setPackage(packageName)
                .putExtra("key", key).putExtra("value", on)
        )
    }

    private fun writeBridge() {
        runCatching {
            File(getExternalFilesDir(null), "ayushmods.json").writeText(
                "{\"headshot\":" + Prefs.on(this, Prefs.HEADSHOT) +
                    ",\"chams\":\"green\",\"chams_on\":" + Prefs.on(this, Prefs.CHAMS) +
                    ",\"anti_ban\":true,\"root\":false,\"rgb\":true,\"build\":\"OB55\",\"v\":\"3.0\"}"
            )
        }
    }

    private fun sync() {
        swHead?.setOn(Prefs.on(this, Prefs.HEADSHOT))
        swChams?.setOn(Prefs.on(this, Prefs.CHAMS))
        reticle?.visibility = if (Prefs.on(this, Prefs.HEADSHOT)) View.VISIBLE else View.GONE
    }

    /* ---------- drag ---------- */
    private fun drag(
        handle: View, root: View, p: WindowManager.LayoutParams,
        onDrop: () -> Unit, onLong: (() -> Unit)?
    ) {
        var sx = 0; var sy = 0; var dx0 = 0f; var dy0 = 0f
        var pending: Runnable? = null

        handle.setOnTouchListener { _, e ->
            when (e.actionMasked) {
                MotionEvent.ACTION_DOWN -> {
                    sx = p.x; sy = p.y; dx0 = e.rawX; dy0 = e.rawY
                    if (onLong != null) {
                        val r = Runnable { onLong(); pending = null }
                        pending = r
                        handle.postDelayed(r, 480L)
                    }
                    true
                }
                MotionEvent.ACTION_MOVE -> {
                    p.x = sx + (e.rawX - dx0).toInt()
                    p.y = sy + (e.rawY - dy0).toInt()
                    runCatching { wm.updateViewLayout(root, p) }
                    if (abs(e.rawX - dx0) > 14f || abs(e.rawY - dy0) > 14f) {
                        pending?.let { handle.removeCallbacks(it); pending = null }
                    }
                    true
                }
                MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> {
                    pending?.let { handle.removeCallbacks(it); pending = null }
                    onDrop()
                    false
                }
                else -> false
            }
        }
    }

    private fun savePos(p: WindowManager.LayoutParams) {
        Prefs.sp(this).edit().putInt(Prefs.X, p.x).putInt(Prefs.Y, p.y).apply()
    }

    /* ---------- foreground ---------- */
    private fun goForeground() {
        val ch = "ayush_panel"
        val nm = getSystemService(NotificationManager::class.java)
        if (nm.getNotificationChannel(ch) == null) {
            nm.createNotificationChannel(
                NotificationChannel(ch, "AYUSH MODS", NotificationManager.IMPORTANCE_LOW)
            )
        }
        val pi = PendingIntent.getActivity(
            this, 0, Intent(this, MainActivity::class.java), PendingIntent.FLAG_IMMUTABLE
        )
        val n: Notification = Notification.Builder(this, ch)
            .setSmallIcon(android.R.drawable.sym_def_app_icon)
            .setContentTitle("AYUSH MODS")
            .setContentText("Premium Panel active · non-root · rgb")
            .setOngoing(true)
            .setContentIntent(pi)
            .build()

        if (Build.VERSION.SDK_INT >= 34)
            startForeground(1, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_SPECIAL_USE)
        else
            startForeground(1, n)
    }

    override fun onDestroy() {
        runCatching { reticle?.let { wm.removeView(it) } }
        runCatching { bubble?.let { wm.removeView(it) } }
        runCatching { panel?.let { wm.removeView(it) } }
        super.onDestroy()
    }

    companion object {
        const val ACTION_TOGGLE = "com.ayush.mods.TOGGLE"
    }
}

class SplashActivity : Activity(), TextToSpeech.OnInitListener {

    private var tts: TextToSpeech? = null
    private var player: MediaPlayer? = null
    private var hint: TextView? = null

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        window.statusBarColor = 0xFF04070A.toInt()
        window.navigationBarColor = 0xFF04070A.toInt()

        val root = LinearLayout(this).apply {
            orientation = LinearLayout.VERTICAL
            gravity = Gravity.CENTER
            setBackgroundColor(0xFF04070A.toInt())
        }

        val logo = RgbBorder(this).apply { cornerDp = 28f }
        logo.addView(
            RgbText(this).apply { text = "A"; textSize = 50f; gravity = Gravity.CENTER },
            FrameLayout.LayoutParams(MATCH, MATCH)
        )
        root.addView(logo, LinearLayout.LayoutParams(dpi(96f), dpi(96f)))

        root.addView(
            RgbText(this).apply {
                text = "AYUSH MODS"
                textSize = 18f
                letterSpacing = 0.28f
                gravity = Gravity.CENTER
            },
            LinearLayout.LayoutParams(WRAP, WRAP).apply { topMargin = dpi(20f) }
        )

        root.addView(
            txt(this, "Premium Panel By Sensei Ayush", 11.5f, C_SUB).apply {
                letterSpacing = 0.1f
                gravity = Gravity.CENTER
            },
            LinearLayout.LayoutParams(WRAP, WRAP).apply { topMargin = dpi(7f) }
        )

        root.addView(
            RgbProgress(this),
            LinearLayout.LayoutParams(dpi(180f), dpi(4f)).apply { topMargin = dpi(30f) }
        )

        hint = txt(this, "initialising rgb engine...", 10f, 0xFF4E6B59.toInt()).apply {
            letterSpacing = 0.1f
        }
        root.addView(hint, LinearLayout.LayoutParams(WRAP, WRAP).apply { topMargin = dpi(14f) })

        setContentView(root)
        root.alpha = 0f
        root.animate().alpha(1f).setDuration(600).start()

        playIntro()
        Handler(Looper.getMainLooper()).postDelayed({ next() }, 4600)
    }

    private fun playIntro() {
        val id = resources.getIdentifier("intro", "raw", packageName)
        if (id != 0) {
            runCatching {
                player = MediaPlayer.create(this, id)
                player?.start()
            }.onFailure { tts = TextToSpeech(this, this) }
        } else {
            tts = TextToSpeech(this, this)
        }
    }

    override fun onInit(status: Int) {
        if (status != TextToSpeech.SUCCESS) return
        val t = tts ?: return
        t.language = Locale.US
        t.setPitch(1.12f)
        t.setSpeechRate(0.80f)
        t.speak(
            "Premium Panel, by Sensei Ayush",
            TextToSpeech.QUEUE_FLUSH,
            Bundle().apply { putFloat(TextToSpeech.Engine.KEY_PARAM_VOLUME, 1f) },
            "intro"
        )
    }

    private fun next() {
        if (hasOverlay()) startPanel() else askOverlay()
    }

    private fun hasOverlay(): Boolean = Settings.canDrawOverlays(this)

    private fun askOverlay() {
        hint?.text = "Allow \"Display over other apps\""
        startActivityForResult(
            Intent(
                Settings.ACTION_MANAGE_OVERLAY_PERMISSION,
                Uri.parse("package:$packageName")
            ), 101
        )
    }

    @Deprecated("legacy result api")
    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        super.onActivityResult(requestCode, resultCode, data)
        if (requestCode != 101) return
        if (hasOverlay()) {
            startPanel()
        } else {
            Toast.makeText(this, "Overlay permission required", Toast.LENGTH_LONG).show()
            Handler(Looper.getMainLooper()).postDelayed({ askOverlay() }, 1200)
        }
    }

    private fun startPanel() {
        hint?.text = "Launching panel..."
        startForegroundService(Intent(this, OverlayService::class.java))
        Handler(Looper.getMainLooper()).postDelayed({
            startActivity(Intent(this, MainActivity::class.java))
            finish()
        }, 750)
    }

    override fun onDestroy() {
        runCatching { player?.release() }
        runCatching { tts?.stop() }
        runCatching { tts?.shutdown() }
        super.onDestroy()
    }
}
